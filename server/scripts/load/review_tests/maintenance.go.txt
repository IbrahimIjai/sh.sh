package maintenance

import (
    "context"
    "testing"
    "time"
)

func TestReviewLateClicksReachRolledReports(t *testing.T) {
    conn := testDatabase(t)
    ctx := context.Background()
    today := dayOf(time.Now())
    m := New(conn, Config{LookbackDays: 3})
    m.now = func() time.Time { return today.Add(12*time.Hour) }
    if _, err := m.Run(ctx, false); err != nil { t.Fatal(err) }
    // A stream consumer recovers a delayed event after its day was already rolled up.
    if _, err := conn.Exec("INSERT INTO clicks (created_at,link_id,visitor_hash) VALUES ($1,1,'late')", today.AddDate(0,0,-2)); err != nil { t.Fatal(err) }
    if _, err := m.Run(ctx, false); err != nil { t.Fatal(err) }
    var n int
    if err := conn.QueryRow("SELECT coalesce(sum(human),0) FROM click_daily WHERE link_id=1").Scan(&n); err != nil { t.Fatal(err) }
    if n != 1 { t.Fatalf("recovered click exists in raw storage but report rollups contain %d, want 1", n) }
}

func TestReviewDefaultPartitionHonorsRetention(t *testing.T) {
    conn := testDatabase(t)
    ctx := context.Background()
    today := dayOf(time.Now())
    if _, err := conn.Exec("INSERT INTO clicks (created_at,link_id) VALUES ($1,1)", today.AddDate(0,0,-40)); err != nil { t.Fatal(err) }
    m := New(conn, Config{RawRetentionDays: 30, LookbackDays: 45})
    if _, err := m.Run(ctx, false); err != nil { t.Fatal(err) }
    var raw, rolled int
    if err := conn.QueryRow("SELECT count(*) FROM clicks_default WHERE created_at < $1", today.AddDate(0,0,-30)).Scan(&raw); err != nil { t.Fatal(err) }
    if err := conn.QueryRow("SELECT coalesce(sum(human),0) FROM click_daily WHERE link_id=1").Scan(&rolled); err != nil { t.Fatal(err) }
    if raw != 0 || rolled != 1 { t.Fatalf("40-day-old click after 30-day retention: raw=%d rolled=%d; want raw=0 rolled=1", raw, rolled) }
}
