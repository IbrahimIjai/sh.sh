package rediscache

import (
    "context"
    "os"
    "testing"
    "time"
)

func TestReviewRedisHonorsRequestDeadline(t *testing.T) {
    url := os.Getenv("TEST_REDIS_URL")
    if url == "" { t.Fatal("isolated TEST_REDIS_URL required") }
    store, err := Open(url)
    if err != nil { t.Fatal(err) }
    defer store.Close()
    // Only pause the runner-owned Redis, before the benchmark API is started.
    if err := store.client.Do(context.Background(), "CLIENT", "PAUSE", "500", "ALL").Err(); err != nil { t.Fatal(err) }
    ctx, cancel := context.WithTimeout(context.Background(), 50*time.Millisecond)
    defer cancel()
    began := time.Now()
    _, _, _, err = store.Get(ctx, "review-deadline-missing")
    elapsed := time.Since(began)
    t.Logf("50ms context completed in %s; error=%v", elapsed, err)
    if elapsed > 200*time.Millisecond { t.Fatalf("Redis I/O ignored the request deadline: %s", elapsed) }
}
